TAM EKRAN ZORLAMASI --- function tamEkranYap() { let elem = document.documentElement; if (elem.requestFullscreen) { elem.requestFullscreen(); } else if (elem.mozRequestFullScreen) { /* Firefox */ elem.mozRequestFullScreen(); } else if (elem.webkitRequestFullscreen) { /* Chrome, Safari & Opera */ elem.webkitRequestFullscreen(); } else if (elem.msRequestFullscreen) { /* IE/Edge */ elem.msRequestFullscreen(); } } // Kullanıcı sayfada herhangi bir yere tıkladığı an tam ekran olur document.addEventListener("click", tamEkranYap); // --- 2. Bu tuşlar burada çalışmaz!"); return false; } // Alt + F4'ü tarayıcı tam engelleyemez ama biz uyaralım if (e.altKey && e.keyCode == 115) { alert("Kaçmaya çalışma!"); return false; } }; // --- 3.
